cmake_minimum_required(VERSION 3.22)
project(AurumWallet VERSION 0.7.2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Widgets Network Test)
find_package(OpenSSL REQUIRED)
find_path(CRYPTOPP_INCLUDE_DIR cryptopp/keccak.h REQUIRED)
find_library(CRYPTOPP_LIBRARY NAMES cryptopp cryptlib REQUIRED)
find_path(QT_WS_INCLUDE_DIR QtWebSockets/QWebSocket PATH_SUFFIXES qt6 REQUIRED)
find_library(QT_WS_LIBRARY NAMES Qt6WebSockets libQt6WebSockets.so.6 Qt6WebSockets.lib REQUIRED)
find_path(SODIUM_INCLUDE_DIR sodium.h REQUIRED)
find_library(SODIUM_LIBRARY NAMES sodium libsodium libsodium.so.26 REQUIRED)
find_path(SECP_INCLUDE_DIR secp256k1.h REQUIRED)
find_library(SECP_LIBRARY NAMES secp256k1 REQUIRED)
find_path(QR_INCLUDE_DIR qrencode.h REQUIRED)
find_library(QR_LIBRARY NAMES qrencode libqrencode libqrencode.so.4 REQUIRED)
add_library(aurum_crypto src/security.cpp src/security.h src/core.cpp src/core.h src/assets.cpp src/evm.cpp src/evm.h src/spl.cpp src/spl.h)
target_include_directories(aurum_crypto PUBLIC src ${SODIUM_INCLUDE_DIR} ${CRYPTOPP_INCLUDE_DIR} ${SECP_INCLUDE_DIR})
target_link_libraries(aurum_crypto PUBLIC Qt6::Core OpenSSL::Crypto ${CRYPTOPP_LIBRARY} ${SODIUM_LIBRARY} ${SECP_LIBRARY})
target_compile_options(aurum_crypto PRIVATE -Wall -Wextra -Wpedantic)
add_library(aurum_core src/network.cpp src/network.h src/market.cpp src/market_history.cpp src/market.h src/electrum.cpp src/electrum.h src/signer.cpp src/signer.h src/monero.cpp src/monero.h src/wallet_profiles.cpp src/wallet_profiles.h src/swap.cpp src/swap.h src/catalog.cpp src/catalog.h)
target_include_directories(aurum_core PUBLIC ${QT_WS_INCLUDE_DIR})
target_link_libraries(aurum_core PUBLIC aurum_crypto Qt6::Network ${QT_WS_LIBRARY})
target_compile_options(aurum_core PRIVATE -Wall -Wextra -Wpedantic)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options(-fstack-protector-strong -fPIE -D_FORTIFY_SOURCE=3)
  target_compile_options(aurum_core PRIVATE -fstack-protector-strong -fPIC -D_FORTIFY_SOURCE=3)
  target_compile_options(aurum_crypto PRIVATE -fstack-protector-strong -fPIC -D_FORTIFY_SOURCE=3)
  # Keep PIE on executable targets only. Applying -pie globally makes the
  # shared hardening preload link as an executable and requires main().
  if(WIN32)
    add_link_options(-Wl,--dynamicbase,--nxcompat,--high-entropy-va)
  else()
    add_link_options(-Wl,-z,relro,-z,now,-z,noexecstack)
  endif()
endif()
add_executable(aurum-signer src/signer_main.cpp)
target_link_libraries(aurum-signer PRIVATE aurum_crypto)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang" AND NOT WIN32)
  target_link_options(aurum-signer PRIVATE -pie)
endif()
if(WIN32)
  # The Linux-only LD_PRELOAD hardener has no Windows equivalent; keep the
  # target so add_dependencies() below stays valid.
  add_custom_target(aurum_monero_hardener)
else()
  add_library(aurum_monero_hardener SHARED src/monero_hardener.cpp)
  set_target_properties(aurum_monero_hardener PROPERTIES PREFIX "lib" OUTPUT_NAME "aurum-monero-hardener")
  target_compile_options(aurum_monero_hardener PRIVATE -fPIC -fvisibility=hidden)
endif()
add_executable(aurum assets.qrc src/main.cpp src/window.cpp src/window_market.cpp src/window_exodus.cpp src/window_startup.cpp src/window_swap.cpp src/mullvad.cpp src/window.h src/mullvad.h)
target_include_directories(aurum PRIVATE ${QR_INCLUDE_DIR})
target_link_libraries(aurum PRIVATE aurum_core Qt6::Widgets ${QR_LIBRARY})
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang" AND NOT WIN32)
  target_link_options(aurum PRIVATE -pie)
endif()
if(WIN32)
  set_target_properties(aurum PROPERTIES WIN32_EXECUTABLE ON)
  target_link_libraries(aurum_crypto PUBLIC ws2_32 crypt32)
endif()
add_dependencies(aurum aurum-signer)
add_dependencies(aurum aurum_monero_hardener)
include(CTest)
if(BUILD_TESTING)
 add_executable(signer_tests tests/signer_tests.cpp)
 target_link_libraries(signer_tests PRIVATE aurum_core Qt6::Test)
 add_dependencies(signer_tests aurum-signer)
 add_test(NAME signer_tests COMMAND signer_tests)
 set_tests_properties(signer_tests PROPERTIES TIMEOUT 120)
 add_executable(security_tests tests/security_tests.cpp)
 target_link_libraries(security_tests PRIVATE aurum_core)
 add_test(NAME security_tests COMMAND security_tests)
 add_executable(chain_probe tests/chain_probe.cpp)
 target_link_libraries(chain_probe PRIVATE aurum_core)
 add_executable(core_tests tests/core_tests.cpp)
 target_link_libraries(core_tests PRIVATE aurum_core Qt6::Test)
 add_test(NAME core_tests COMMAND core_tests)
endif()
install(TARGETS aurum aurum-signer RUNTIME DESTINATION bin)
if(NOT WIN32)
  install(TARGETS aurum_monero_hardener LIBRARY DESTINATION bin)
endif()
if(EXISTS "${CMAKE_SOURCE_DIR}/third_party/monero/monero-wallet-rpc")
  install(PROGRAMS third_party/monero/monero-wallet-rpc
          DESTINATION bin)
  if(EXISTS "${CMAKE_SOURCE_DIR}/third_party/monero/monerod")
    install(PROGRAMS third_party/monero/monerod
            DESTINATION bin)
  endif()
  install(FILES third_party/monero/LICENSE third_party/monero/SOURCE.txt
          DESTINATION share/aurum/monero)
endif()
if(BUILD_TESTING)
  add_executable(ui_tests assets.qrc tests/ui_tests.cpp src/window.cpp src/window_market.cpp src/window_exodus.cpp src/window_startup.cpp src/window_swap.cpp src/mullvad.cpp src/window.h src/mullvad.h)
  target_include_directories(ui_tests PRIVATE ${QR_INCLUDE_DIR})
  target_link_libraries(ui_tests PRIVATE aurum_core Qt6::Widgets Qt6::Test ${QR_LIBRARY})
  add_dependencies(ui_tests aurum-signer)
  add_dependencies(ui_tests aurum_monero_hardener)
  add_test(NAME ui_tests COMMAND ui_tests)
  set_tests_properties(ui_tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 45)
endif()
set_target_properties(aurum aurum-signer PROPERTIES INSTALL_RPATH "$ORIGIN/../lib")
if(BUILD_TESTING)
 add_executable(electrum_tests tests/electrum_tests.cpp)
 target_link_libraries(electrum_tests PRIVATE aurum_core Qt6::Test)
 add_test(NAME electrum_tests COMMAND electrum_tests)
 set_tests_properties(electrum_tests PROPERTIES TIMEOUT 45)
endif()
if(BUILD_TESTING)
 add_executable(market_tests tests/market_tests.cpp)
 target_link_libraries(market_tests PRIVATE aurum_core Qt6::Test)
 add_test(NAME market_tests COMMAND market_tests)
 set_tests_properties(market_tests PROPERTIES TIMEOUT 30)
endif()
if(BUILD_TESTING)
 add_executable(monero_tests tests/monero_tests.cpp)
 target_link_libraries(monero_tests PRIVATE aurum_core Qt6::Test)
 add_dependencies(monero_tests aurum_monero_hardener)
 add_test(NAME monero_tests COMMAND monero_tests)
 set_tests_properties(monero_tests PROPERTIES TIMEOUT 40)
endif()
if(BUILD_TESTING)
 add_executable(wallet_profiles_tests tests/wallet_profiles_tests.cpp)
 target_link_libraries(wallet_profiles_tests PRIVATE aurum_core Qt6::Test)
 add_dependencies(wallet_profiles_tests aurum-signer)
 add_test(NAME wallet_profiles_tests COMMAND wallet_profiles_tests)
 set_tests_properties(wallet_profiles_tests PROPERTIES TIMEOUT 45)
endif()
